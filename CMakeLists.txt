cmake_minimum_required(VERSION 3.20)
project(PhoneMic VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RECEIVER_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(RECEIVER_DEFAULT ON)
endif()
option(PHONEMIC_BUILD_RECEIVER "Build the Linux PipeWire receiver" ${RECEIVER_DEFAULT})
find_package(Threads REQUIRED)
enable_testing()
add_executable(core-tests linux/tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE shared)
target_link_libraries(core-tests PRIVATE Threads::Threads)
target_compile_options(core-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME core-tests COMMAND core-tests)
add_executable(config-tests linux/tests/config_tests.cpp)
target_include_directories(config-tests PRIVATE linux/src)
target_compile_options(config-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME config-tests COMMAND config-tests)
if(PHONEMIC_BUILD_RECEIVER)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.49)
  add_executable(phonemic linux/src/main.cpp)
  target_include_directories(phonemic PRIVATE shared)
  target_link_libraries(phonemic PRIVATE PkgConfig::PIPEWIRE Threads::Threads)
  target_compile_options(phonemic PRIVATE -Wall -Wextra)
  include(GNUInstallDirs)
  install(TARGETS phonemic RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
  set(PHONEMIC_EXECUTABLE "\"${CMAKE_INSTALL_FULL_BINDIR}/phonemic\"")
  set(PHONEMIC_CONFIG "\"%E/phonemic/receiver.conf\"")
  configure_file(linux/systemd/phonemic.service.in phonemic.service @ONLY)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/phonemic.service" DESTINATION lib/systemd/user)
  install(FILES linux/receiver.conf.example DESTINATION share/doc/phonemic)
endif()
